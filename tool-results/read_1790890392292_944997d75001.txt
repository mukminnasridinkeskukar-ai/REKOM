   155→-- 8. TRIGGER: buat profile otomatis saat user mendaftar (auth.users -> profiles)
   156→-- ============================================================================
   157→create or replace function public.handle_new_user()
   158→returns trigger
   159→language plpgsql
   160→security definer
   161→set search_path = public
   162→as $$
   163→begin
   164→  insert into public.profiles (id, email, nama_lengkap, nik, no_hp, asal_instansi, jabatan, role)
   165→  values (
   166→    new.id,
   167→    new.email,
   168→    coalesce(new.raw_user_meta_data ->> 'nama_lengkap', split_part(coalesce(new.email, 'Pemohon'), '@', 1)),
   169→    nullif(new.raw_user_meta_data ->> 'nik', ''),
   170→    nullif(new.raw_user_meta_data ->> 'no_hp', ''),
   171→    coalesce(nullif(new.raw_user_meta_data ->> 'asal_instansi', ''), 'Pemohon'),
   172→    nullif(new.raw_user_meta_data ->> 'jabatan', ''),
   173→    'pemohon' -- role default; Super Admin dapat mengubahnya dari menu Kelola Pengguna
   174→  );
   175→  return new;
   176→end;
   177→$$;
   178→
   179→drop trigger if exists on_auth_user_created on auth.users;
   180→create trigger on_auth_user_created
   181→  after insert on auth.users
   182→  for each row execute function public.handle_new_user();

   201→-- boleh melihat pengajuan? (pemilik / verifikator bidang / manajemen)
   202→create or replace function public.can_view_pengajuan(p_pengajuan_id uuid)
   203→returns boolean
   204→language sql stable security definer set search_path = public
   205→as $$
   206→  select exists (
   207→    select 1
   208→    from public.pengajuan_rekom p
   209→    join public.master_jenis_rekom j on j.id = p.jenis_id
   210→    where p.id = p_pengajuan_id
   211→      and (
   212→        p.pemohon_id = auth.uid()
   213→        or (public.my_role() = 'verifikator_bidang' and j.bidang = public.my_bidang())
   214→        or public.my_role() in ('admin_tu','kabid','kadis','super_admin')
   215→      )
   216→  );
   217→$$;

   219→-- ============================================================================
   220→-- 10. RPC: GENERATE NOMOR SURAT (format: 440/{seq}/{kode}/Dinkes-Kukar/{year})
   221→-- ============================================================================
   222→create or replace function public.generate_nomor_surat(p_jenis_id uuid)
   223→returns text
   224→language plpgsql
   225→security definer
   226→set search_path = public
   227→as $$
   228→declare
   229→  v_kode     text;
   230→  v_template text;
   231→  v_seq      int;
   232→  v_year     int := extract(year from now())::int;
   233→begin
   234→  select kode_jenis, template_nomor into v_kode, v_template
   235→  from public.master_jenis_rekom where id = p_jenis_id;
   236→  if v_kode is null then
   237→    raise exception 'Jenis rekomendasi tidak ditemukan';
   238→  end if;
   239→
   240→  select count(*) + 1 into v_seq
   241→  from public.pengajuan_rekom
   242→  where jenis_id = p_jenis_id
   243→    and nomor_surat is not null
   244→    and extract(year from coalesce(tgl_terbit, created_at))::int = v_year;
   245→
   246→  return replace(replace(replace(v_template,
   247→    '{seq}',  lpad(v_seq::text, 3, '0')),
   248→    '{kode}', v_kode),
   249→    '{year}', v_year::text);
   250→end;
   251→$$;

   253→-- ============================================================================
   254→-- 11. RPC: VERIFIKASI PUBLIK VIA QR (tanpa login — hanya data aman)
   255→-- ============================================================================
   256→create or replace function public.verify_surat_public(p_qr text)
   257→returns table (
   258→  nomor_surat     text,
   259→  jenis_nama      text,
   260→  judul_pengajuan text,
   261→  pemohon_nama    text,
   262→  asal_instansi   text,
   263→  tgl_terbit      timestamptz,
   264→  nama_kadis      text,
   265→  pengajuan_id    uuid
   266→)
   267→language plpgsql
   268→stable
   269→security definer
   270→set search_path = public
   271→as $$
   272→begin
   273→  return query
   274→  select p.nomor_surat,
   275→         j.nama_jenis,
   276→         p.judul_pengajuan,
   277→         pr.nama_lengkap,
   278→         pr.asal_instansi,
   279→         p.tgl_terbit,
   280→         (select pk.nama_lengkap from public.profiles pk where pk.role = 'kadis' limit 1) as nama_kadis,
   281→         p.id
   282→  from public.pengajuan_rekom p
   283→  join public.master_jenis_rekom j on j.id = p.jenis_id
   284→  join public.profiles pr on pr.id = p.pemohon_id
   285→  where p.qr_code_id::text = p_qr
   286→    and p.status = 'terbit';
   287→end;
   288→$$;
