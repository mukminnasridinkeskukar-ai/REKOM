   202→   202→create or replace function public.can_view_pengajuan(p_pengajuan_id uuid)
   203→   203→returns boolean
   204→   204→language sql stable security definer set search_path = public
   205→   205→as $$
   206→   206→  select exists (
   207→   207→    select 1
   208→   208→    from public.pengajuan_rekom p
   209→   209→    join public.master_jenis_rekom j on j.id = p.jenis_id
   210→   210→    where p.id = p_pengajuan_id
   211→   211→      and (
   212→   212→        p.pemohon_id = auth.uid()
   213→   213→        or (public.my_role() = 'verifikator_bidang' and j.bidang = public.my_bidang())
   214→   214→        or public.my_role() in ('admin_tu','kabid','kadis','super_admin')
   215→   215→      )
   216→   216→  );
   217→   217→$$;

   220→   220→-- 10. RPC: GENERATE NOMOR SURAT (format: 440/{seq}/{kode}/Dinkes-Kukar/{year})
   221→   221→-- ============================================================================
   222→   222→create or replace function public.generate_nomor_surat(p_jenis_id uuid)
   223→   223→returns text
   224→   224→language plpgsql
   225→   225→security definer
   226→   226→set search_path = public
   227→   227→as $$
   228→   228→declare
   229→   229→  v_kode     text;
   230→   230→  v_template text;
   231→   231→  v_seq      int;
   232→   232→  v_year     int := extract(year from now())::int;
   233→   233→begin
   234→   234→  select kode_jenis, template_nomor into v_kode, v_template
   235→   235→  from public.master_jenis_rekom where id = p_jenis_id;
   236→   236→  if v_kode is null then
   237→   237→    raise exception 'Jenis rekomendasi tidak ditemukan';
   238→   238→  end if;
   239→   239→
   240→   240→  select count(*) + 1 into v_seq
   241→   241→  from public.pengajuan_rekom
   242→   242→  where jenis_id = p_jenis_id
   243→   243→    and nomor_surat is not null
   244→   244→    and extract(year from coalesce(tgl_terbit, created_at))::int = v_year;
   245→   245→
   246→   246→  return replace(replace(replace(v_template,
   247→   247→    '{seq}',  lpad(v_seq::text, 3, '0')),
   248→   248→    '{kode}', v_kode),
   249→   249→    '{year}', v_year::text);
   250→   250→end;
   251→   251→$$;

   254→   254→-- 11. RPC: VERIFIKASI PUBLIK VIA QR (tanpa login — hanya data aman)
   255→   255→-- ============================================================================
   256→   256→create or replace function public.verify_surat_public(p_qr text)
   257→   257→returns table (
   258→   258→  nomor_surat     text,
   259→   259→  jenis_nama      text,
   260→   260→  judul_pengajuan text,
   261→   261→  pemohon_nama    text,
   262→   262→  asal_instansi   text,
   263→   263→  tgl_terbit      timestamptz,
   264→   264→  nama_kadis      text,
   265→   265→  pengajuan_id    uuid
   266→   266→)
   267→   267→language plpgsql
   268→   268→stable
   269→   269→security definer
   270→   270→set search_path = public
   271→   271→as $$
   272→   272→begin
   273→   273→  return query
   274→   274→  select p.nomor_surat,
   275→   275→         j.nama_jenis,
   276→   276→         p.judul_pengajuan,
   277→   277→         pr.nama_lengkap,
   278→   278→         pr.asal_instansi,
   279→   279→         p.tgl_terbit,
   280→   280→         (select pk.nama_lengkap from public.profiles pk where pk.role = 'kadis' limit 1) as nama_kadis,
   281→   281→         p.id
   282→   282→  from public.pengajuan_rekom p
   283→   283→  join public.master_jenis_rekom j on j.id = p.jenis_id
   284→   284→  join public.profiles pr on pr.id = p.pemohon_id
   285→   285→  where p.qr_code_id::text = p_qr
   286→   286→    and p.status = 'terbit';
   287→   287→end;
   288→   288→$$;
